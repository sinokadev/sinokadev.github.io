  <p id="countdown-detail">2027년 1월 1일까지</p>

<style>
  .countdown {
    margin: 2rem 0;
    padding: 2rem 1rem;
    text-align: center;
  }
  .countdown-number {
    margin: 0;
    font-size: clamp(3rem, 14vw, 7rem);
    font-weight: 800;
    line-height: 1.1;
    font-variant-numeric: tabular-nums;
  }
  #countdown-detail { margin: 0.75rem 0 0; }
</style>

<script>
  const countdownNumber = document.getElementById('countdown-number');
  const countdownDetail = document.getElementById('countdown-detail');
  // 한국 표준시 기준 2027년 1월 1일 00:00
  const newYear = Date.UTC(2026, 11, 31, 15, 0, 0);

  function updateCountdown() {
    const remaining = newYear - Date.now();
    if (remaining > 0) {
      const days = Math.ceil(remaining / 86_400_000);
      countdownNumber.textContent = days === 1 ? 'D-1' : `D-${days}`;
      countdownDetail.textContent = `2027년 1월 1일까지 ${days}일`;
    } else if (remaining > -1_000) {
      countdownNumber.textContent = 'D-DAY';
      countdownDetail.textContent = '2027년이 시작됐습니다!';
    } else {
      const days = Math.floor(-remaining / 86_400_000);
      countdownNumber.textContent = `D+${days}`;
      countdownDetail.textContent = `2027년이 시작된 지 ${days}일`;
    }
  }

  updateCountdown();
  window.setInterval(updateCountdown, 1_000);
</script>
