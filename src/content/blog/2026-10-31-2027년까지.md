<script>
  const countdownNumber = document.getElementById('countdown-number');
  const countdownTime = document.getElementById('countdown-time');
  // 한국 표준시 기준 2027년 1월 1일 00:00
  const newYear = Date.UTC(2026, 11, 31, 15, 0, 0);

  function updateCountdown() {
    const now = Date.now();
    const remaining = newYear - now;
    const kstToday = new Date(now + 9 * 60 * 60 * 1_000);
    const todayAtUtc = Date.UTC(
      kstToday.getUTCFullYear(),
      kstToday.getUTCMonth(),
      kstToday.getUTCDate()
    );
    const targetDate = Date.UTC(2027, 0, 1);
    const calendarDays = Math.round((targetDate - todayAtUtc) / 86_400_000);

    if (calendarDays > 0) {
      countdownNumber.textContent = `D-${calendarDays}`;
    } else if (calendarDays === 0) {
      countdownNumber.textContent = 'D-DAY';
    } else {
      countdownNumber.textContent = `D+${Math.abs(calendarDays)}`;
    }

    const duration = Math.abs(remaining);
    const days = Math.floor(duration / 86_400_000);
    const hours = Math.floor((duration % 86_400_000) / 3_600_000);
    const minutes = Math.floor((duration % 3_600_000) / 60_000);
    const seconds = Math.floor((duration % 60_000) / 1_000);
    const clock = [hours, minutes, seconds]
      .map((part) => String(part).padStart(2, '0'))
      .join(':');

    countdownTime.textContent = remaining >= 0
      ? `남은 시간 ${days}일 ${clock}`
      : `지난 시간 ${days}일 ${clock}`;
  }

  updateCountdown();
  window.setInterval(updateCountdown, 1_000);
</script>
